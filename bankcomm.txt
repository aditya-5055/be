pragma solidity ^0.8.20;
// Specifies the Solidity compiler version.
// ^0.8.20 means version 0.8.20 or compatible newer versions below 0.9.0.

contract BankAccount {
    // A contract is similar to a class in C++.
    // BankAccount is the name of our smart contract.


    // MAPPING:
    // A mapping stores data in key-value pairs.
    //
    // Here:
    // address = Key (user's Ethereum wallet address)
    // uint256 = Value (user's balance)
    //
    // Each wallet address has its own balance.
    //
    // private means the mapping cannot be accessed directly
    // from outside this contract.

    mapping(address => uint256) private balances;


    // DEPOSIT FUNCTION:
    // This function allows users to deposit ETH into the contract.
    
    function deposit() public payable {

        // public means anyone can call this function.
        // payable allows this function to receive ETH.
        //
        // msg.value = Amount of ETH sent with this transaction.
        // msg.sender = Wallet address of the person calling the function.
        //
        // require() checks a condition.
        // If the condition is false, the transaction is reverted.
        //
        // Here, we check that the deposit amount is greater than zero.

        require(msg.value > 0, "Deposit amount must be greater than zero");

        // Add the deposited ETH to the balance of the caller.
        //
        // balances[msg.sender] gets the caller's current balance.
        // += adds the deposited amount to the existing balance.
        //
        // Example:
        // Previous balance = 2 ETH
        // Deposit amount   = 1 ETH
        // New balance      = 3 ETH

        balances[msg.sender] += msg.value;
    }


    // WITHDRAW FUNCTION:
    // This function allows users to withdraw ETH from their balance.
    //
    // amount = How much ETH the user wants to withdraw.
    // IMPORTANT: amount must be provided in Wei, not ETH.

    function withdraw(uint256 amount) public {

        // STEP 1: CHECK WITHDRAWAL AMOUNT
        // The user cannot withdraw zero or a negative amount.
        // uint256 cannot store negative numbers.

        require(amount > 0, "Amount must be greater than zero");


        // STEP 2: CHECK AVAILABLE BALANCE
        // Check whether the user's balance is enough for withdrawal.
        //
        // Example:
        // User balance = 2 ETH
        // Withdrawal   = 3 ETH
        // Condition becomes false, so the transaction is reverted.

        require(balances[msg.sender] >= amount, "Insufficient balance");


        // STEP 3: UPDATE USER BALANCE
        // Deduct the withdrawal amount from the user's balance.
        //
        // Example:
        // Previous balance = 3 ETH
        // Withdrawal       = 1 ETH
        // Remaining balance = 2 ETH
        //
        // We update the balance BEFORE transferring ETH.
        // This helps protect against reentrancy attacks.

        balances[msg.sender] -= amount;


        // STEP 4: TRANSFER ETH TO USER
        //
        // payable(msg.sender) converts the user's address
        // into a payable address so it can receive ETH.
        //
        // call{value: amount}("") sends the specified amount of ETH.
        //
        // call returns two values:
        // bool success = Whether the transfer was successful.
        // bytes data   = Returned data from the external call.
        //
        // We use a blank space after the comma because
        // we do not need the returned data.
        
        (bool success, ) = payable(msg.sender).call{value: amount}("");


        // STEP 5: CHECK TRANSFER SUCCESS
        // If the ETH transfer fails, success will be false.
        // require() will revert the complete transaction.
        // This also restores the balance deduction.

        require(success, "Transfer failed");
    }


    // GET BALANCE FUNCTION:
    // This function displays the balance of the wallet
    // that is calling this function.

    function getBalance() public view returns (uint256) {

        // public means anyone can call this function.
        //
        // view means the function only reads blockchain data.
        // It does not modify the stored balance.
        //
        // returns (uint256) means this function returns a number.
        //
        // balances[msg.sender] gets the balance of the caller.
        //
        // The returned balance is in Wei.

        return balances[msg.sender];
    }
}
